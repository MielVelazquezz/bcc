É uma árvore que quando insiro o elemento, se o elemento que insere a deixa desbalanceada, o algoritmo corrige, deixando a árvore balanceada.

o fator de balanceamento é dito pela **altura**: FB{-1, 0, 1}


### Tipo 1: (2)(1) ou (-2)(-1)

**Exemplo 1:** 

```C
void RotacaoEsq(Tree **p){
	Tree *q, *temp;
	q = (*p)->dir;
	temp = q->esq;
	
	q->esq = *p;
	(*p)->dir = temp;
	*p = q;
}
```

```C
void RotacaoDir(Tree **p){
	Tree *q, *temp;
	q = (*p)->esq;
	temp = q->dir;
	
	q->dir = *p;
	(*p)->esq = temp;
	*p = q;
}
```

### Tipo 2: (2)(-1) ou (-2)(1)

**Exemplo 1:**

```C
void InsereAVL(Tree **raiz, int info, char *flag){
	int FB_pai, FB_filho;
	
	if(*raiz == NULL){
		*raiz = CriaNo(info);
		*flag = 1;
	}
	else{
		if(info<(*raiz)->info)
			InsereAVL((*raiz)->esq, info, &*flag);
		else
			InsereAVL((*raiz)->dir, info, &*flag);
	}
	
	if(*flag == 1){
		FB_pai = altura((*raiz)->dir) - altura((*raiz)->esq);
		if(FB_pai == 2 || -2){ //desbalanceado
			*flag = 0;
			if(FB_pai == 2){
				FB_filho = altura((*raiz)->dir->dir) - altura((*raiz)->dir->esq);
				if(FB_filho == 1){ // (2)(1)
					RotacaoEsq(&*raiz);
				}
				else{ // (2)(-1)
					RotacaoDir(&(*raiz)->dir);
					RotacaoEsq(&*raiz);
				}
			}
			else{ // FB_pai == -2
				FB_filho = altura((*raiz)->esq->dir) - altura((*raiz)->esq->esq);
				if(FB_filho == -1){ // (-2)(-1)
					RotacaoDir(&*raiz);
				}
				else{ // (-2)(1)
					RotacaoEsq(&*(raiz)->esq);
					RotacaoDir(&*raiz);
				}
			}
		}
		else{
			
		}
	}

}
```
