Dois tipos de nodos:
-  **Terminais**: nodos que armazenam os atomos contidos na lista generalizada.
-  **Nao-terminais**: nodos que descrevem a forma da lista, isto é, a estrutura de aninhamento e relacionamento dos átomos nela contidos.

![[Pasted image 20260827154529.png]]

Nodos não-terminais são compostos por dois campos, um que armazena um ponteiro para baixo (cabeça) e outro que armazena um ponteiro para frente (cauda)

### <mark style="background: #FFF3A3A6;">Operações Básicas de Listas Generalizadas</mark>

**Cria átomo:**
```C
ListaGen CriaT(char info){
	ListaGen = (ListaGen *)malloc(sizeof(ListaGen));
	L->terminal = 1;
	strcpy(L->no.info, info);
	return L;
}
```

**Verificar se está vazia:**
```C
char Nula(ListaGen *L){
	return L==NULL;
}
```

**Verificar se é átomo:**
```C
char Atomo(ListaGen *L){
	return !Nula(L) && L->terminal);
}
```

**Constrói a lista:**
```C
ListaGen *Cons(ListaGen *H, ListaGen *T){
	if(Atomo(T)){
		return NULL;
	}
	ListaGen *L = (ListaGen*)malloc(sizeof(ListaGen))
	L->terminal=0;
	L->no.lista.cabeca = H;
	L->no.lista.cauda = T;
	return L;
}
```

**Pega a cabeça da lista:**
```C
ListaGen *Head(ListaGen *L){
	if(Nula(L)||Atomo(L)){
		printf("Head: argumento deve ser lista nao vazia!\n");
		return NULL;
	}
	return L->no.lista.cabeca; 
}
```

**Pega o rabo kk:**
```C
ListaGen *Tail(ListaGen *L){
	if(!Nula(L)||Atomo(L)){
		printf("Tail: argumento deve ser lista nao vazia!\n");
		return NULL;
	}
	return L->no.lista.cauda;
}
```

### <mark style="background: #FFF3A3A6;">Algoritmos para exibir uma lista generalizada</mark>
Duas possibilidades:
- Aponta pra um atomo
- Aponta pra uma lista generalizada

**1. Recursivo com Interativo** 
``` C
void Exibe(ListaGen *L){
	if(Atomo(L))
		printf("%s", L->no.info);
	else{
		printf("[");
		
		while(!Nula(L)){
			Exibe(Head(L));
			L=Tail(L);
			
			if(!Nula(L))
				printf(",");
		}
		
		printf("]");
	}
}
```


**2. Recursivo**
```C
void ExibeAtomo(ListaGen *L){
	if(!Nula(L)){
		if(Atomo(L))
			printf("%s", L->no.info);
		else{
			ExibeAtomo(Head(L));
			ExibeAtomo(Tail(L));
		}
	}
}
```

### <mark style="background: #FFF3A3A6;">Algoritmo para destruir uma lista generalizada</mark>
São três casos a considerar:
- L tem valor `NULL`
- L aponta um nodo terminal;
- L aponta um nodo não terminal


```C
void Destruir(ListaGen **L){
	if(!Nula(*L)){
		if(Atomo(*L)) free (*L);
	}
	else{
		Destruir(&(*L)->no.lista.cabeca);
		Destruir(&(*L)->no.lista.cauda);
		free (*L);
		(*L)=NULL;
	}
}
```

### <mark style="background: #FFF3A3A6;">Algoritmo para contar quantidade de listas</mark>

```C
int ContListas(ListaGen *L){
	int cont=0;
	Fila *F;
	init(&F);
	enqueue(&F, L);
	while(!isEmpty(F)){
		dequeue(&F, &L);
		cont++;
		while(!Nula(L)){
			if(!Nula(Head(L))&&!Atomo(Head(L)))
				enqueue(&F, Head(L));
			L=Tail(L);	
		}
	}
	
	return cont;
}
```

### Algoritmo para comparar duas listas generalizadas

```C
char CompararListas(ListaGen *L1, ListaGen *L2){
	if(Nula(L1)&&Nula(L2)) return 1;
	if(Nula(L1)||Nula(L2)) return 0;	
	if(Atomo(L1)&&Atomo(L2)) return !strcmp(L1->no.info, L2->no.info);
	if(Atomo(L1)||Atomo(L2)) return 0;
	return CompararListas(Head(L1), Head(L2)) && CompararListas(Tail(L1), Tail(L2));
}
```

| Notação         | O que significa                                      | Quando usar                                                                                      |
| --------------- | ---------------------------------------------------- | ------------------------------------------------------------------------------------------------ |
| `struct X *p`   | `p` guarda o endereço de um `struct X`               | Parâmetro normal, percorrer lista                                                                |
| `*p`            | "o conteúdo apontado por `p`" (dereferência)         | Ler/escrever o valor apontado, ex: `*p = 10`                                                     |
| `p->campo`      | Atalho pra `(*p).campo`                              | Sempre que `p` é ponteiro pra struct — é o que você mais vai usar                                |
| `struct X **pp` | `pp` guarda o endereço de **um ponteiro**            | Só se você insistir em alterar o ponteiro-pai por referência (dá pra evitar, como mostrei acima) |
| `&x`            | Endereço de `x`                                      | Pra pegar o endereço de uma variável comum ou struct (não-ponteiro)                              |
| `&*p`           | Você quase nunca precisa disso — `&*p` é igual a `p` | Ignora essa combinação, ela "se cancela"                                                         |
> [!tip] Lembrando 
> **se a variável já é ponteiro, você usa `->` pra acessar campos. Se não é ponteiro, usa `.`**. E só usa `&` quando precisa do endereço de algo que "não é naturalmente" um ponteiro (variável local, struct por valor).
